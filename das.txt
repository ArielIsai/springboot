    const API_URL = 'http://localhost:8080/api';

    async function cargarDatos() {
        try {
            const resDepto = await fetch(`${API_URL}/departamentos`);
            const departamentos = await resDepto.json();

            const tbodyDepto = document.querySelector('#tablaDepartamentos tbody') || document.querySelectorAll('table')[0].querySelector('tbody');
            const selectDepto = document.getElementById('selectDepto');
            const filtroDepto = document.getElementById('filtroDepto');

            tbodyDepto.innerHTML = '';
            selectDepto.innerHTML = '<option value="">Seleccione Departamento</option>';

                selectDepto.innerHTML += `<option value="${d.id}">${d.nombre}</option>`;
                filtroDepto.innerHTML += `<option value="${d.id}">${d.nombre}</option>`;
            });

            cargarEmpleados(`${API_URL}/empleados`);
        } catch (error) {
            console.error("Error al cargar datos:", error);
        }
    }

    async function cargarEmpleados(url) {
        const resEmp = await fetch(url);
        const empleados = await resEmp.json();
        const tables = document.querySelectorAll('table');
        const tbodyEmp = tables[1].querySelector('tbody');
        tbodyEmp.innerHTML = '';

        empleados.forEach(e => {
            tbodyEmp.innerHTML += `<tr>
                    <td>${e.id}</td>
                    <td>${e.nombre}</td>
                    <td>${e.apellido}</td>
                    <td>${e.cedula}</td>
                    <td>${e.departamento ? e.departamento.nombre : 'Sin depto'}</td>
                </tr>`;
        });
    }

            body: JSON.stringify({ nombre: document.getElementById('nombreDepto').value })
        });
        document.getElementById('nombreDepto').value = '';

    const API_URL = 'http://localhost:8080/api';
    let todosLosLibros = [];

    async function cargarDatos() {
        try {
            const resAutor = await fetch(`${API_URL}/autores`);
            const autores = await resAutor.json();

            const tbodyAutor = document.getElementById('tablaAutores');
            const selectAutor = document.getElementById('selectAutor');

            tbodyAutor.innerHTML = '';
            selectAutor.innerHTML = '<option value="">Seleccione Autor</option>';

            autores.forEach(a => {
                tbodyAutor.innerHTML += `<tr><td>${a.id}</td><td>${a.nombre}</td></tr>`;
                selectAutor.innerHTML += `<option value="${a.id}">${a.nombre}</option>`;
            });

            await cargarLibros();
        } catch (error) {
            console.error("Error al cargar datos:", error);
        }
    }

    async function cargarLibros(nombreFiltro = '') {
        try {
            const resLibro = await fetch(`${API_URL}/libros`);
            todosLosLibros = await resLibro.json();

            let librosAMostrar = todosLosLibros;
            if (nombreFiltro.trim() !== '') {
                const filtroLower = nombreFiltro.toLowerCase();
                librosAMostrar = todosLosLibros.filter(l => l.autor && l.autor.nombre.toLowerCase().includes(filtroLower));
            }

            const tbodyLibro = document.getElementById('tablaLibros');
            tbodyLibro.innerHTML = '';

            librosAMostrar.forEach(l => {
                tbodyLibro.innerHTML += `<tr>
                    <td>${l.id}</td>
                    <td>${l.titulo}</td>
                    <td>${l.autor ? l.autor.nombre : 'Sin autor'}</td>
                    <td><button onclick="eliminarLibro(${l.id})">Eliminar</button></td>
                </tr>`;
            });
        } catch (error) {
            console.error("Error al cargar libros:", error);
        }
    }

    document.getElementById('formAutor').addEventListener('submit', async (e) => {
        e.preventDefault();
        await fetch(`${API_URL}/autores`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ nombre: document.getElementById('nombreAutor').value })
        });
        e.target.reset();
        cargarDatos();
    });

    document.getElementById('formLibro').addEventListener('submit', async (e) => {
        e.preventDefault();
        const libro = {
            titulo: document.getElementById('tituloLibro').value,
            autor: { id: parseInt(document.getElementById('selectAutor').value) }
        };
        await fetch(`${API_URL}/libros`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(libro)
        });
        e.target.reset();
        cargarDatos();
    });

    document.getElementById('filtroAutor').addEventListener('input', (e) => {
        const nombreAutor = e.target.value;
        cargarLibros(nombreAutor);
    });

    async function eliminarLibro(id) {
        if (confirm('¿Deseas eliminar este libro?')) {
            await fetch(`${API_URL}/libros/${id}`, { method: 'DELETE' });
            cargarDatos();
        }
    }

    cargarDatos();
