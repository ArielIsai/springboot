import jakarta.persistence.*;

@Entity
@Table(name = "cuentas")
public class Cuenta {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, unique = true)
    private String numeroCuenta;

    @Column(nullable = false)
    private String titular;

    @Column(nullable = false)
    private Double saldo = 0.0;

    public Cuenta() {}

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }

    public String getNumeroCuenta() { return numeroCuenta; }
    public void setNumeroCuenta(String numeroCuenta) { this.numeroCuenta = numeroCuenta; }

    public String getTitular() { return titular; }
    public void setTitular(String titular) { this.titular = titular; }

    public Double getSaldo() { return saldo; }
    public void setSaldo(Double saldo) { this.saldo = saldo; }
}

package com.example.banco.model;

import jakarta.persistence.*;
import java.time.LocalDateTime;

@Entity
@Table(name = "movimientos")
public class Movimiento {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false)
    private String tipo; // "DEPOSITO" o "RETIRO"

    @Column(nullable = false)
    private Double monto;

    private LocalDateTime fecha = LocalDateTime.now();

    @ManyToOne
    @JoinColumn(name = "cuenta_id", nullable = false)
    private Cuenta cuenta;

    public Movimiento() {}

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }

    public String getTipo() { return tipo; }
    public void setTipo(String tipo) { this.tipo = tipo; }

    public Double getMonto() { return monto; }
    public void setMonto(Double monto) { this.monto = monto; }

    public LocalDateTime getFecha() { return fecha; }
    public void setFecha(LocalDateTime fecha) { this.fecha = fecha; }

    public Cuenta getCuenta() { return cuenta; }
    public void setCuenta(Cuenta cuenta) { this.cuenta = cuenta; }
}

package com.example.banco.controller;

import com.example.banco.model.Cuenta;
import com.example.banco.model.Movimiento;
import com.example.banco.repository.CuentaRepository;
import com.example.banco.repository.MovimientoRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.time.LocalDateTime;
import java.util.List;

@RestController
@RequestMapping("/api/movimientos")
@CrossOrigin(origins = "*")
public class MovimientoController {

    @Autowired
    private MovimientoRepository movimientoRepository;

    @Autowired
    private CuentaRepository cuentaRepository;

    @GetMapping
    public List<Movimiento> listar() {
        return movimientoRepository.findAll();
    }

    @PostMapping
    public ResponseEntity<?> registrar(@RequestBody Movimiento movimiento) {
        Cuenta cuenta = cuentaRepository.findById(movimiento.getCuenta().getId()).orElse(null);

        if (cuenta == null) {
            return ResponseEntity.badRequest().body("Cuenta no encontrada");
        }

        if (movimiento.getMonto() == null || movimiento.getMonto() <= 0) {
            return ResponseEntity.badRequest().body("Monto inválido");
        }

        if ("DEPOSITO".equalsIgnoreCase(movimiento.getTipo())) {
            cuenta.setSaldo(cuenta.getSaldo() + movimiento.getMonto());
        } else if ("RETIRO".equalsIgnoreCase(movimiento.getTipo())) {
            if (cuenta.getSaldo() < movimiento.getMonto()) {
                return ResponseEntity.badRequest().body("Saldo insuficiente");
            }
            cuenta.setSaldo(cuenta.getSaldo() - movimiento.getMonto());
        } else {
            return ResponseEntity.badRequest().body("Tipo de movimiento inválido");
        }

        cuentaRepository.save(cuenta);
        movimiento.setCuenta(cuenta);
        movimiento.setFecha(LocalDateTime.now());
        Movimiento guardado = movimientoRepository.save(movimiento);

        return ResponseEntity.ok(guardado);
    }
}

    const API_URL = 'http://localhost:8080/api';
    let todosLosLibros = [];

    async function cargarDatos() {
        try {
            const resAutor = await fetch(`${API_URL}/autores`);
            const autores = await resAutor.json();

            const tbodyAutor = document.getElementById('tablaAutores');
            const selectAutor = document.getElementById('selectAutor');

            tbodyAutor.innerHTML = '';
            selectAutor.innerHTML = '<option value="">Seleccione Autor</option>';

            autores.forEach(a => {
                tbodyAutor.innerHTML += `<tr><td>${a.id}</td><td>${a.nombre}</td></tr>`;
                selectAutor.innerHTML += `<option value="${a.id}">${a.nombre}</option>`;
            });

            await cargarLibros();
        } catch (error) {
            console.error("Error al cargar datos:", error);
        }
    }

    async function cargarLibros(nombreFiltro = '') {
        try {
            const resLibro = await fetch(`${API_URL}/libros`);
            todosLosLibros = await resLibro.json();

            let librosAMostrar = todosLosLibros;
            if (nombreFiltro.trim() !== '') {
                const filtroLower = nombreFiltro.toLowerCase();
                librosAMostrar = todosLosLibros.filter(l => l.autor && l.autor.nombre.toLowerCase().includes(filtroLower));
            }

            const tbodyLibro = document.getElementById('tablaLibros');
            tbodyLibro.innerHTML = '';

            librosAMostrar.forEach(l => {
                tbodyLibro.innerHTML += `<tr>
                    <td>${l.id}</td>
                    <td>${l.titulo}</td>
                    <td>${l.autor ? l.autor.nombre : 'Sin autor'}</td>
                    <td><button onclick="eliminarLibro(${l.id})">Eliminar</button></td>
                </tr>`;
            });
        } catch (error) {
            console.error("Error al cargar libros:", error);
        }
    }

    document.getElementById('formAutor').addEventListener('submit', async (e) => {
        e.preventDefault();
        await fetch(`${API_URL}/autores`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ nombre: document.getElementById('nombreAutor').value })
        });
        e.target.reset();
        cargarDatos();
    });

    document.getElementById('formLibro').addEventListener('submit', async (e) => {
        e.preventDefault();
        const libro = {
            titulo: document.getElementById('tituloLibro').value,
            autor: { id: parseInt(document.getElementById('selectAutor').value) }
        };
        await fetch(`${API_URL}/libros`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(libro)
        });
        e.target.reset();
        cargarDatos();
    });

    document.getElementById('filtroAutor').addEventListener('input', (e) => {
        const nombreAutor = e.target.value;
        cargarLibros(nombreAutor);
    });

    async function eliminarLibro(id) {
        if (confirm('¿Deseas eliminar este libro?')) {
            await fetch(`${API_URL}/libros/${id}`, { method: 'DELETE' });
            cargarDatos();
        }
    }

    cargarDatos();

    const API_URL = 'http://localhost:8080/api';
    let listaEmpleadosGlobal = [];

    async function cargarDatos() {
        try {
            const resDepto = await fetch(`${API_URL}/departamentos`);
            const departamentos = await resDepto.json();

            const tbodyDepto = document.querySelectorAll('table')[0].querySelector('tbody');
            const selectDepto = document.getElementById('selectDepto');

            tbodyDepto.innerHTML = '';
            selectDepto.innerHTML = '<option value="">Seleccione Departamento</option>';

            departamentos.forEach(d => {
                tbodyDepto.innerHTML += `<tr><td>${d.id}</td><td>${d.nombre}</td></tr>`;
                selectDepto.innerHTML += `<option value="${d.id}">${d.nombre}</option>`;
            });

            await cargarEmpleados();
        } catch (error) {
            console.error("Error al cargar datos:", error);
        }
    }

    async function cargarEmpleados() {
        const resEmp = await fetch(`${API_URL}/empleados`);
        listaEmpleadosGlobal = await resEmp.json();
        mostrarEmpleados(listaEmpleadosGlobal);
    }

    function mostrarEmpleados(empleados) {
        const tables = document.querySelectorAll('table');
        const tbodyEmp = tables[1].querySelector('tbody');
        tbodyEmp.innerHTML = '';

        empleados.forEach(e => {
            tbodyEmp.innerHTML += `<tr>
                    <td>${e.id}</td>
                    <td>${e.nombre}</td>
                    <td>${e.departamento ? e.departamento.nombre : 'Sin depto'}</td>
                </tr>`;
        });
    }

    document.getElementById('formDepartamento').addEventListener('submit', async (event) => {
        event.preventDefault();

        const departamento = {
            id: 0,
            nombre: document.getElementById('nombreDepto').value
        };

        const response = await fetch(`${API_URL}/departamentos`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(departamento)
        });

        if (response.ok) {
            document.getElementById('nombreDepto').value = '';
            cargarDatos();
        } else {
            console.error("Error al guardar departamento");
        }
    });

    document.getElementById('formEmpleado').addEventListener('submit', async (event) => {
        event.preventDefault();

        const empleado = {
            id: 0,
            nombre: document.getElementById('nombreEmp').value,
            departamento: {
                id: parseInt(document.getElementById('selectDepto').value),
                nombre: ""
            }
        };

        const response = await fetch(`${API_URL}/empleados`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(empleado)
        });

        if (response.ok) {
            event.target.reset();
            cargarDatos();
        } else {
            console.error("Error al guardar empleado");
        }
    });

    document.getElementById('inputFiltroDepto').addEventListener('input', (e) => {
        const textoFiltro = e.target.value.toLowerCase();

            if (!emp.departamento) return false;
            return emp.departamento.nombre.toLowerCase().includes(textoFiltro);
        });

        mostrarEmpleados(empleadosFiltrados);
    });

    cargarDatos();

    <input type="text" id="tituloLibro" placeholder="Título del Libro" required>
    <select id="selectAutor" required>
        <option value="">Seleccione Autor</option>
    </select>
    <button type="submit">Guardar Libro</button>
</form>
<br>

<label>Filtrar libros por Autor:</label>
<input type="text" id="filtroAutor" placeholder="Escribe el nombre del autor...">
<br><br>

<table border="1">
    <thead>
    <tr>
        <th>ID</th>
        <th>Título</th>
        <th>Autor</th>
    </tr>
    </thead>
    <tbody id="tablaLibros"></tbody>
</table>

<script>
    const API_URL = 'http://localhost:8080/api';
    let todosLosLibros = [];

    async function cargarDatos() {
        try {
            const resAutor = await fetch(`${API_URL}/autores`);
            const autores = await resAutor.json();

            const tbodyAutor = document.getElementById('tablaAutores');
            const selectAutor = document.getElementById('selectAutor');

            tbodyAutor.innerHTML = '';
            selectAutor.innerHTML = '<option value="">Seleccione Autor</option>';

            autores.forEach(a => {
                tbodyAutor.innerHTML += `<tr><td>${a.id}</td><td>${a.nombre}</td></tr>`;
                selectAutor.innerHTML += `<option value="${a.id}">${a.nombre}</option>`;
            });

            await cargarLibros();
        } catch (error) {
            console.error("Error al cargar datos:", error);
        }
    }

    async function cargarLibros(nombreFiltro = '') {
        try {
            const resLibro = await fetch(`${API_URL}/libros`);
            todosLosLibros = await resLibro.json();

            let librosAMostrar = todosLosLibros;
            if (nombreFiltro.trim() !== '') {
                const filtroLower = nombreFiltro.toLowerCase();
                librosAMostrar = todosLosLibros.filter(l => l.autor && l.autor.nombre.toLowerCase().includes(filtroLower));
            }

            const tbodyLibro = document.getElementById('tablaLibros');
            tbodyLibro.innerHTML = '';

            librosAMostrar.forEach(l => {
                tbodyLibro.innerHTML += `<tr>
                    <td>${l.id}</td>
                    <td>${l.titulo}</td>
                    <td>${l.autor ? l.autor.nombre : 'Sin autor'}</td>
                </tr>`;
            });
        } catch (error) {
            console.error("Error al cargar libros:", error);
        }
    }

    document.getElementById('formAutor').addEventListener('submit', async (e) => {
        e.preventDefault();
        const autor = {
            nombre: document.getElementById('nombreAutor').value
        };
        const res = await fetch(`${API_URL}/autores`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(autor)
        });
        if (res.ok) {
            document.getElementById('formAutor').reset();
            cargarDatos();
        } else {
            console.error("Error al guardar autor");
        }
    });

    document.getElementById('formLibro').addEventListener('submit', async (e) => {
        e.preventDefault();
        const libro = {
            titulo: document.getElementById('tituloLibro').value,
            autor: {
                id: parseInt(document.getElementById('selectAutor').value)
            }
        };
        const res = await fetch(`${API_URL}/libros`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(libro)
        });
        if (res.ok) {
            document.getElementById('formLibro').reset();
            cargarDatos();
        } else {
            console.error("Error al guardar libro");
        }
    });

    document.getElementById('filtroAutor').addEventListener('input', (e) => {
        const nombreAutor = e.target.value;
        cargarLibros(nombreAutor);
    });

    cargarDatos();
</script>
</body>
</html>
